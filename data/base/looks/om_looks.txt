# OpenMerc GFX-modern r_om_look: the colour of the picture (gl3_look.c)
#
# The picture is graded in float at the end of the view; SoF's 8-bit colour table stays neutral. Edit this file
# (a copy in the game folder, openmerc/looks/om_looks.txt, overrides the one in the pak) and type
# r_om_looks_reload in the console. r_om_look_strength 0..1 (menu: Gloom) blends from neutral to the look;
# r_om_look_preset <name> forces a look on every map; r_om_look 0 brings back the original colour table
# (renderer restart).
#
# look <name> <ev> <toe> <shoulder> <saturation> <shadow r g b> <highlight r g b> <vignette> <grain> <adapt> [lut]
#   ev          exposure in stops (-0.5 = a third darker)
#   toe         presses the blacks down, mid grey stays (0 none, 0.4 deep)
#   shoulder    where the highlights start to roll off towards white (0.6 = the original curve)
#   saturation  1 = as is, 0.7 = muted
#   shadow      tint of the dark tones, highlight: tint of the light (luminance is kept; 1 1 1 = none)
#   vignette    darkening towards the corners (0..1), grain: film grain (0.03 = fine)
#   adapt       share of the automatic exposure (1 = full; 0.5 lets dark rooms stay dark)
#   lut         optional strip of N slices of N x N (looks/neutral_lut.png graded in an image editor)
# map <map name, '*' allowed> <look>      the first matching line wins
